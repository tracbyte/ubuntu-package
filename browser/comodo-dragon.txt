install:
========
sudo apt update && sudo apt install comodo-dragon -y

upgrade:
========
sudo apt update && sudo apt install --only-upgrade comodo-dragon -y

uninstall:
==========
sudo apt remove --purge comodo-dragon -y

clean cache:
============
sudo apt clean && sudo apt autoremove -y

remove profiles:
================
rm -rf ~/.config/comodo-dragon ~/.cache/comodo-dragon ~/.local/share/comodo-dragon

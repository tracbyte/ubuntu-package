install:
========
# Update package lists and install required prerequisites
sudo apt update && sudo apt install -y curl apt-transport-https ca-certificates gnupg

# Add Pale Moon's signing key
curl -fsSL https://download.opensuse.org/repositories/home:/stevenpusser/xUbuntu_22.04/Release.key | sudo gpg --dearmor -o /usr/share/keyrings/palemoon.gpg

# Add the official Pale Moon repository
echo "deb [signed-by=/usr/share/keyrings/palemoon.gpg] http://download.opensuse.org/repositories/home:/stevenpusser/xUbuntu_22.04/ /" | sudo tee /etc/apt/sources.list.d/palemoon.list

# Update package lists with the new repository
sudo apt update

# Install Pale Moon
sudo apt install -y palemoon

update:
=======
# Fetch latest repository indexes and upgrade Pale Moon only
sudo apt update && sudo apt --only-upgrade install -y palemoon

clean cache:
============
# Remove Pale Moon local cache files
rm -rf ~/.cache/moonchild\ productions/pale\ moon

remove profiles:
================
# Remove Pale Moon user profile data and settings
rm -rf ~/.moonchild\ productions/pale\ moon

uninstall:
==========
# Remove Pale Moon and purge its configuration files
sudo apt remove --purge -y palemoon

# Remove the Pale Moon repository source file
sudo rm -f /etc/apt/sources.list.d/palemoon.list

# Remove the saved Pale Moon signing key
sudo rm -f /usr/share/keyrings/palemoon.gpg

install:
========
# Update package lists and install required prerequisites
sudo apt update && sudo apt install -y curl apt-transport-https ca-certificates gnupg

# Add Vivaldi's signing key
curl -fSsL https://repo.vivaldi.com/archive/linux_signing_key.pub | sudo gpg --dearmor -o /usr/share/keyrings/vivaldi-browser.gpg

# Add the official Vivaldi repository
echo "deb [signed-by=/usr/share/keyrings/vivaldi-browser.gpg] https://repo.vivaldi.com/archive/deb/ stable main" | sudo tee /etc/apt/sources.list.d/vivaldi.list

# Update package lists with the new repository
sudo apt update

# Install Vivaldi
sudo apt install -y vivaldi-stable

update:
=======
# Fetch latest repository indexes and upgrade Vivaldi only
sudo apt update && sudo apt --only-upgrade install -y vivaldi-stable

clean cache:
============
# Remove Vivaldi local cache files
rm -rf ~/.cache/vivaldi

remove profiles:
================
# Remove Vivaldi user profile data and settings
rm -rf ~/.config/vivaldi

uninstall:
==========
# Remove Vivaldi and purge its configuration files
sudo apt remove --purge -y vivaldi-stable

# Remove the Vivaldi repository source file
sudo rm -f /etc/apt/sources.list.d/vivaldi.list

# Remove the saved Vivaldi signing key
sudo rm -f /usr/share/keyrings/vivaldi-browser.gpg

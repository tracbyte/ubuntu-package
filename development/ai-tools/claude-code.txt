install:
========
# Update system packages, install Node.js environment, and deploy Claude Code globally
sudo apt update && sudo apt install nodejs npm -y && sudo npm install -g @anthropic-ai/claude-code

update:
=======
# Fetch and install the newest global update for Claude Code
sudo npm update -g @anthropic-ai/claude-code

clean cache:
============
# Wipe out Anthropic CLI cache indexes
rm -rf ~/.cache/claude-code

remove profiles:
================
# Wipe out Anthropic CLI local authentication profiles and history logs
rm -rf ~/.config/claude-code

uninstall:
==========
# Uninstall the global Claude Code package from the Node module registry
sudo npm uninstall -g @anthropic-ai/claude-code

install:
========
# Update system tools and install Node.js environment to deploy the Anthropic Claude SDK
sudo apt update && sudo apt install nodejs npm -y && npm install @anthropic-ai/sdk

update:
=======
# Upgrade local Anthropic Claude project dependency library
npm update @anthropic-ai/sdk

clean cache:
============
# Remove Node package manager local cache files
npm cache clean --force

remove profiles:
================
# Remove custom local project configuration scripts or keys linked to Claude
rm -rf ~/.config/claude

uninstall:
==========
# Remove the Anthropic Claude SDK local module
npm uninstall @anthropic-ai/sdk

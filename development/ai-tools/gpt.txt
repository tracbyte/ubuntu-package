install:
========
# Update package files, install python pipelines, and deploy Shell GPT via pip
sudo apt update && sudo apt install python3-pip -y && pip3 install sgpt --break-system-packages

update:
=======
# Pull the latest release of the Shell GPT command-line tool via pip
pip3 install --upgrade sgpt --break-system-packages

clean cache:
============
# Remove Shell GPT local cache settings
rm -rf ~/.cache/shell-gpt

remove profiles:
================
# Wipe out custom OpenAI API keys and local prompt histories
rm -rf ~/.config/shell-gpt

uninstall:
==========
# Uninstall the Shell GPT core package from the global pip container
pip3 uninstall sgpt shell-gpt -y --break-system-packages

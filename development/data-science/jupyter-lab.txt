install:
========
# Update package list, install Python environment tools, and install JupyterLab
sudo apt update && sudo apt install -y python3-pip python3-venv
pip3 install jupyterlab --break-system-packages

# To open Jupyter Lab use any one of these cmd
jupyter-lab 

OR

jupyter lab

update:
=======
# Check and upgrade JupyterLab package via pip manager
pip3 install --upgrade jupyterlab --break-system-packages

clean cache:
============
# Remove JupyterLab local build caches and staging directories
rm -rf ~/.cache/jupyter ~/.local/share/jupyter/lab

remove profiles:
================
# Remove user configuration files, workspaces, and preferences
rm -rf ~/.jupyter

uninstall:
==========
# Remove JupyterLab package via pip manager
pip3 uninstall -y jupyterlab --break-system-packages

# Remove user configuration files, workspaces, and local build cache
rm -rf ~/.jupyter ~/.cache/jupyter ~/.local/share/jupyter/lab

install:
======
# Install Visual Studio Code via Snap with classic confinement
sudo snap install --classic code

update:
=======
# Refresh the Visual Studio Code snap package to the latest release
sudo snap refresh code

clean cache:
==========
# Remove workspace cache and temporary application data
rm -rf ~/.config/Code/Cache ~/.config/Code/CachedData

remove profiles:
==============
# Remove user settings, keybindings, and extension profiles
rm -rf ~/.config/Code ~/.vscode

uninstall:
========
# Remove Visual Studio Code snap package and its snap data
sudo snap remove code

# Remove remaining user configuration and extension directories
rm -rf ~/.config/Code ~/.vscode

install:
========
sudo apt update && sudo apt install dotnet-sdk-8.0 -y

upgrade:
========
sudo apt update && sudo apt install --only-upgrade dotnet-sdk-8.0 -y

uninstall:
==========
sudo apt remove --purge dotnet-sdk-8.0 -y

clean cache:
============
sudo apt clean && sudo apt autoremove -y

remove profiles:
================
rm -rf ~/.config/dotnet-sdk-8.0 ~/.cache/dotnet-sdk-8.0 ~/.local/share/dotnet-sdk-8.0

install:
========
# Update the package list and install the GNU compiler collection, debugger, and build automation frameworks
sudo apt update && sudo apt install gcc g++ gdb make cmake -y

update:
=======
# Refresh system repository listings and upgrade the GCC and G++ compilers alongside make tools
sudo apt update && sudo apt install --only-upgrade gcc g++ gdb make cmake -y

clean cache:
============
# Remove local GDB history and temporary build artifacts
rm -f ~/.gdb_history

remove profiles:
================
# Remove user profile build environment settings
rm -rf ~/.cmake

uninstall:
==========
# Uninstall the compilers, system debuggers, and build-essential configuration files
sudo apt purge gcc g++ gdb make cmake build-essential -y

# Clean up leftover system package dependencies and runtimes
sudo apt autoremove -y

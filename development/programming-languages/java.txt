install:
========
# Update the package list and install the latest official LTS OpenJDK development kit and runtime environment
sudo apt update && sudo apt install default-jdk default-jre -y

update:
=======
# Refresh system repository listings and upgrade the OpenJDK development kit and runtime
sudo apt update && sudo apt install --only-upgrade default-jdk default-jre -y

clean cache:
============
# Wipe out certificate runtime storage keys and temporary Java cache
rm -rf ~/.java/.userPrefs

remove profiles:
================
# Wipe out hidden Java user preferences and profiles
rm -rf ~/.java

uninstall:
==========
# Uninstall all OpenJDK development kits, runtimes, and system configurations
sudo apt purge default-jdk default-jre openjdk* -y

# Clean up leftover system package dependencies
sudo apt autoremove -y

install:
========
# Update package lists and install the PHP interpreter, common CLI extensions, and package manager
sudo apt update && sudo apt install php php-cli php-common php-mbstring php-xml php-zip php-curl curl -y

update:
=======
# Refresh system repository listings and upgrade PHP along with its core extensions
sudo apt update && sudo apt install --only-upgrade php php-cli php-common -y

clean cache:
============
# Wipe out local Composer caches
rm -rf ~/.composer/cache

remove profiles:
================
# Wipe out Composer profiles, configurations, and system settings
rm -rf ~/.composer ~/.config/php

uninstall:
==========
# Uninstall PHP packages, module configurations, and CLI dependencies
sudo apt purge php* -y

# Clean up leftover system package dependencies
sudo apt autoremove -y

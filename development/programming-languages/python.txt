install:
========
# Update package lists and install the full Python 3 development environment, compiler tools, and pip manager
sudo apt update && sudo apt install python3 python3-pip python3-dev python3-venv build-essential -y

update:
=======
# Refresh system repository listings and upgrade Python 3 development tools and libraries
sudo apt update && sudo apt install --only-upgrade python3 python3-pip python3-dev python3-venv -y

clean cache:
============
# Clear local pip cache files
rm -rf ~/.cache/pip

remove profiles:
================
# Wipe out Python history log and user-wide pip libraries/profiles
rm -rf ~/.python_history ~/.local/lib/python*

uninstall:
==========
# Uninstall pip, virtual environment packages, and development configurations
sudo apt purge python3-pip python3-dev python3-venv build-essential -y

# Clean up leftover system package dependencies
sudo apt autoremove -y

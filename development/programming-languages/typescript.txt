install:
========
# Install TypeScript globally using Node's npm package manager
sudo npm install -g typescript

update:
=======
# Upgrade the globally installed TypeScript compiler to the latest version
sudo npm install -g typescript@latest

clean cache:
============
# Wipe out local TypeScript cache
rm -rf ~/.cache/typescript

remove profiles:
================
# Remove global TypeScript compiler preferences and configuration stores
rm -rf ~/.config/typescript

uninstall:
==========
# Uninstall global TypeScript binaries and compiler libraries
sudo npm uninstall -g typescript

install:
========
# Update package list, install prerequisites, and configure the official GitHub CLI repository
sudo apt update && sudo apt install curl -y
curl -fsSL https://cli.github.com/packages/githubcli-archive-keyring.gpg | sudo dd of=/usr/share/keyrings/githubcli-archive-keyring.gpg > /dev/null
sudo chmod go+r /usr/share/keyrings/githubcli-archive-keyring.gpg
echo "deb [arch=$(dpkg --print-architecture) signed-by=/usr/share/keyrings/githubcli-archive-keyring.gpg] https://cli.github.com/packages stable main" | sudo tee /etc/apt/sources.list.d/github-cli.list > /dev/null
sudo apt update && sudo apt install gh -y

update:
=======
# Refresh system repository listings and upgrade the GitHub CLI tool
sudo apt update && sudo apt install --only-upgrade gh -y

clean cache:
============
# Remove GitHub CLI local cache files
rm -rf ~/.cache/gh

remove profiles:
================
# Wipe out stored GitHub CLI authentication tokens and configuration settings
rm -rf ~/.config/gh

uninstall:
==========
# Uninstall the GitHub CLI package and system configurations
sudo apt purge gh -y

# Remove the repository setup and signing keyring
sudo rm -f /etc/apt/sources.list.d/github-cli.list
sudo rm -f /usr/share/keyrings/githubcli-archive-keyring.gpg

# Clean up leftover system package dependencies
sudo apt autoremove -y

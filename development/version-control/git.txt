install:
========
# Update your system package index and install Git with its core configuration tools
sudo apt update && sudo apt install git build-essential -y

update:
=======
# Refresh system repository listings and upgrade Git to the latest stable system release
sudo apt update && sudo apt install --only-upgrade git -y

clean cache:
============
# Clean local Git cache directories
rm -rf ~/.cache/git

remove profiles:
================
# Wipe out global git configurations, user identity settings, and credential storage keys
rm -f ~/.gitconfig
rm -rf ~/.config/git

uninstall:
==========
# Uninstall the Git application software and its global system package configurations
sudo apt purge git -y

# Clean up leftover system package dependencies and libraries
sudo apt autoremove -y

install:
========
sudo apt update && sudo apt install uget -y

upgrade:
========
sudo apt update && sudo apt install --only-upgrade uget -y

uninstall:
==========
sudo apt remove --purge uget -y

clean cache:
============
sudo apt clean && sudo apt autoremove -y

remove profiles:
================
rm -rf ~/.config/uget ~/.cache/uget ~/.local/share/uget

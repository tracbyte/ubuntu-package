install:
========
sudo apt update && sudo apt install vlc -y

upgrade:
========
sudo apt update && sudo apt install --only-upgrade vlc -y

uninstall:
==========
sudo apt remove --purge vlc -y

clean cache:
============
sudo apt clean && sudo apt autoremove -y

remove profiles:
================
rm -rf ~/.config/vlc ~/.cache/vlc ~/.local/share/vlc

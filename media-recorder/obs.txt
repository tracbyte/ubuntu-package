install:
========
sudo apt update && sudo apt install obs-studio -y

upgrade:
========
sudo apt update && sudo apt install --only-upgrade obs-studio -y

uninstall:
==========
sudo apt remove --purge obs-studio -y

clean cache:
============
sudo apt clean && sudo apt autoremove -y

remove profiles:
================
rm -rf ~/.config/obs-studio ~/.cache/obs-studio ~/.local/share/obs-studio

install:
========
sudo apt update && sudo apt install openoffice -y

upgrade:
========
sudo apt update && sudo apt install --only-upgrade openoffice -y

uninstall:
==========
sudo apt remove --purge openoffice -y

clean cache:
============
sudo apt clean && sudo apt autoremove -y

remove profiles:
================
rm -rf ~/.config/openoffice ~/.cache/openoffice ~/.local/share/openoffice
